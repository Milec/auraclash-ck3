# 0001 Mortal Core Training menu
# 0002 scheduled auto-training check (hidden)
# 0100 choose Martial Arts Style
# 0200/0201 choose Element (two pages)
# 0300/0301 choose Foundation (two pages)
# 0400 Preserved Youth health check (hidden)

ac_cult.0001 = {
	type = character_event
	title = ac_cult.0001.t
	desc = {
		desc = ac_cult.0001.desc
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = ac_core_training_cooldown }
				desc = ac_cult.0001.desc_cooldown
			}
			desc = ac_cult.0001.desc_ready
		}
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = ac_core_training_auto }
				desc = ac_cult.0001.desc_auto_on
			}
			desc = ac_cult.0001.desc_auto_off
		}
	}
	theme = martial
	left_portrait = {
		character = root
		animation = personality_bold
	}

	option = { # Soul Core -> learning
		name = ac_cult.0001.soul
		trigger = { ac_can_train_core_trigger = { CORE = soul } }
		show_as_unavailable = { always = yes }
		ac_train_core_effect = { CORE = soul }
		ai_chance = {
			base = 10
			modifier = {
				add = 20
				learning >= 12
			}
		}
	}

	option = { # Agility Core -> intrigue
		name = ac_cult.0001.agility
		trigger = { ac_can_train_core_trigger = { CORE = agility } }
		show_as_unavailable = { always = yes }
		ac_train_core_effect = { CORE = agility }
		ai_chance = {
			base = 10
			modifier = {
				add = 20
				intrigue >= 12
			}
		}
	}

	option = { # Power Core -> martial
		name = ac_cult.0001.power
		trigger = { ac_can_train_core_trigger = { CORE = power } }
		show_as_unavailable = { always = yes }
		ac_train_core_effect = { CORE = power }
		ai_chance = {
			base = 10
			modifier = {
				add = 20
				martial >= 12
			}
		}
	}

	option = { # Fortitude Core -> stewardship
		name = ac_cult.0001.fortitude
		trigger = { ac_can_train_core_trigger = { CORE = fortitude } }
		show_as_unavailable = { always = yes }
		ac_train_core_effect = { CORE = fortitude }
		ai_chance = {
			base = 10
			modifier = {
				add = 20
				stewardship >= 12
			}
		}
	}

	option = { # turn auto-training on
		name = ac_cult.0001.auto_on
		trigger = { NOT = { has_character_flag = ac_core_training_auto } }
		add_character_flag = ac_core_training_auto
		custom_tooltip = ac_cult.0001.auto_on.tt
		hidden_effect = { trigger_event = ac_cult.0001 }
		ai_chance = { base = 0 }
	}

	option = { # turn auto-training off
		name = ac_cult.0001.auto_off
		trigger = { has_character_flag = ac_core_training_auto }
		remove_character_flag = ac_core_training_auto
		custom_tooltip = ac_cult.0001.auto_off.tt
		ai_chance = { base = 0 }
	}

	option = {
		name = ac_cult.0001.leave
		ai_chance = { base = 1 }
	}
}

ac_cult.0400 = {
	type = character_event
	hidden = yes

	trigger = {
		is_alive = yes
		has_trait = ac_preserved_youth
	}

	immediate = {
		remove_character_flag = ac_health_watch
		if = {
			limit = { health < 3 }
			random = {
				chance = ac_low_health_death_chance
				ac_low_health_death_effect = yes
			}
		}
		if = {
			limit = { is_alive = yes }
			add_character_flag = {
				flag = ac_health_watch
				days = 100
			}
			if = {
				limit = { health < 3 }
				trigger_event = {
					id = ac_cult.0400
					days = 30
				}
			}
			else = {
				trigger_event = {
					id = ac_cult.0400
					days = 90
				}
			}
		}
	}
}
