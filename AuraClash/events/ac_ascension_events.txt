ac_ascend.0050 = {
	type = character_event
	title = ac_ascend.0050.t
	desc = {
		desc = ac_ascend.0050.desc_scores
		first_valid = {
			triggered_desc = {
				trigger = { ac_ascension_total >= ac_ascension_top_total }
				desc = ac_ascend.0050.desc_top
			}
			triggered_desc = {
				trigger = { ac_ascension_total >= ac_ascension_pass_total }
				desc = ac_ascend.0050.desc_admitted
			}
			desc = ac_ascend.0050.desc_eliminated
		}
		first_valid = {
			triggered_desc = {
				trigger = { exists = var:ac_ascension_rival ac_ascension_total > var:ac_ascension_rival.ac_ascension_expected_total }
				desc = ac_ascend.0050.desc_beat_rival
			}
			triggered_desc = {
				trigger = { exists = var:ac_ascension_rival }
				desc = ac_ascend.0050.desc_lost_rival
			}
		}
		desc = ac_ascend.0050.desc_favor
	}
	theme = martial
	left_portrait = { character = root animation = personality_honorable }
	trigger = { is_alive = yes has_character_flag = ac_ascension_entrant }

	option = {
		name = ac_ascend.0050.ok
		if = {
			limit = { ac_ascension_total >= ac_ascension_pass_total }
			ac_join_archonate_effect = yes
			if = {
				limit = { ac_ascension_total >= ac_ascension_top_total }
				add_prestige = 300
				custom_tooltip = ac_ascend.0050.top_tt
			}
		}
		else = {
			add_prestige = 50
			ac_gain_aura_pressure_effect = { AP = 10 }
		}
		hidden_effect = { ac_ascension_finish_effect = yes }
	}
}
