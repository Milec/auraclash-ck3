ac_ascend.0050 = {
	type = character_event
	title = ac_ascend.0050.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_character_flag = ac_ascension_duel_won ac_ascension_score_value >= ac_ascension_admission_score }
				desc = ac_ascend.0050.desc_admitted
			}
			triggered_desc = {
				trigger = { ac_ascension_score_value >= ac_ascension_mention_score }
				desc = ac_ascend.0050.desc_mention
			}
			desc = ac_ascend.0050.desc_eliminated
		}
	}
	theme = martial
	left_portrait = { character = root animation = personality_honorable }
	trigger = { is_alive = yes has_character_flag = ac_ascension_entrant }

	option = {
		name = ac_ascend.0050.ok
		if = {
			limit = { has_character_flag = ac_ascension_duel_won ac_ascension_score_value >= ac_ascension_admission_score }
			ac_join_archonate_effect = yes
		}
		else_if = {
			limit = { ac_ascension_score_value >= ac_ascension_mention_score }
			add_prestige = 300
			ac_add_random_core_dot_effect = yes
		}
		else = { add_prestige = 50 }
		hidden_effect = { ac_ascension_finish_effect = yes }
	}
}
