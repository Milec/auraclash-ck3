ac_sect_test_01_initialization = {
	title_target = k_england
	assert = { has_variable = ac_kingdom_sect }
	expect = {
		var:ac_kingdom_sect = {
			exists = var:ac_headquarters
			var:ac_kingdom = root
			var:ac_upper_count = 8
			var:ac_lower_count = 16
			story_owner = { has_trait = ac_sect_master has_trait = ac_cultivator }
		}
	}
	effect = {
		var:ac_kingdom_sect = { save_scope_as = ac_original_sect }
		save_scope_as = ac_sect_kingdom
		title_capital_county.title_province = { save_scope_as = ac_sect_hq }
		holder = {
			create_character = { gender_female_chance = 50 age = 30 culture = this.culture faith = this.faith location = scope:ac_sect_hq save_scope_as = ac_test_duplicate }
		}
		scope:ac_test_duplicate = { ac_sect_found_effect = yes }
	}
	expect = {
		var:ac_kingdom_sect = scope:ac_original_sect
		scope:ac_test_duplicate = { NOT = { has_variable = ac_sect } }
	}
}

ac_sect_test_06_relationships_and_succession = {
	title_target = k_england
	assert = { has_variable = ac_kingdom_sect title:k_scotland = { exists = holder } }
	effect = {
		var:ac_kingdom_sect = { save_scope_as = ac_inherited_sect story_owner = { save_scope_as = actor } }
		title:k_scotland.var:ac_kingdom_sect.story_owner = { save_scope_as = recipient }
		ac_sect_alliance_effect = yes
		var:ac_kingdom_sect = { save_scope_as = ac_join_sect }
		title:k_scotland.holder = {
			save_scope_as = ac_test_ruler
			ac_sect_join_effect = yes
			remove_trait = ac_sect_lower_disciple
			add_trait = ac_sect_upper_disciple
			set_variable = { name = ac_sect_merit value = 50000 }
		}
		var:ac_kingdom_sect = {
			random_in_list = {
				variable = ac_members
				limit = { has_trait = ac_sect_lower_disciple }
				set_variable = { name = ac_sect_merit value = 100000 }
			}
			story_owner = { death = natural }
		}
	}
	expect = {
		var:ac_kingdom_sect = scope:ac_inherited_sect
		var:ac_kingdom_sect.story_owner = scope:ac_test_ruler
		var:ac_kingdom_sect.var:ac_master = scope:ac_test_ruler
		var:ac_kingdom_sect.var:ac_alchemy_pavilion = 1
		var:ac_kingdom_sect = { is_target_in_variable_list = { name = ac_allies target = scope:recipient.var:ac_sect } }
		scope:recipient.var:ac_sect = { is_target_in_variable_list = { name = ac_allies target = scope:ac_inherited_sect } }
		scope:ac_test_ruler = { has_trait = ac_sect_master primary_title = title:k_scotland }
	}
}

ac_sect_test_10_native_trial_admission_and_promotion = {
	title_target = k_england
	assert = { any_activity = { has_activity_type = activity_ac_sect_trials var:ac_trial_sect ?= title:k_england.var:ac_kingdom_sect } }
	effect = {
		random_activity = {
			limit = { has_activity_type = activity_ac_sect_trials var:ac_trial_sect ?= title:k_england.var:ac_kingdom_sect }
			save_scope_as = activity
			activity_host = { save_scope_as = host }
		}
		holder = {
			create_character = {
				gender_female_chance = 50
				age = 25
				culture = this.culture faith = this.faith location = scope:activity.activity_location
				save_scope_as = ac_test_examinee
			}
		}
		scope:ac_test_examinee = {
			add_to_activity_without_travel = scope:activity
			ac_sect_begin_trial_effect = yes
			set_variable = { name = ac_sect_trial_score value = 100 }
			ac_sect_complete_trial_effect = yes
		}
	}
	expect = { scope:ac_test_examinee = { involved_activity = scope:activity has_trait = ac_cultivator has_trait = ac_sect_lower_disciple var:ac_sect = title:k_england.var:ac_kingdom_sect } }
	effect = {
		scope:ac_test_examinee = {
			remove_character_flag = ac_sect_trial_cooldown
			set_variable = { name = ac_sect_contribution value = 200 }
			set_variable = { name = ac_sect_promotion_requested value = 1 }
			ac_sect_begin_trial_effect = yes
		}
	}
	assert = { scope:ac_test_examinee = { var:ac_sect_contribution = 50 has_variable = ac_sect_trial_paid } }
	effect = {
		scope:ac_test_examinee = {
			set_variable = { name = ac_sect_trial_score value = 100 }
			ac_sect_complete_trial_effect = yes
			ac_sect_complete_trial_effect = yes
		}
	}
	expect = {
		scope:ac_test_examinee = {
			has_trait = ac_sect_upper_disciple NOT = { has_trait = ac_sect_lower_disciple }
			var:ac_sect_contribution = 50 NOT = { has_variable = ac_sect_trial_paid }
			ac_element_count = 2
		}
	}
}
