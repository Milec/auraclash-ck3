# Portrait age for cultivators. The slowed phase starts when the character is both a
# cultivator and at least 22 (var:ac_youth_start_age), from their apparent age then
# (var:ac_youth_start_visual): they age 5 visible years over as many
# real years as their longevity bonus, then age normally from there.
ac_youth_end_age = {
	value = var:ac_youth_start_age
	add = ac_longevity_bonus
}
ac_visual_age = {
	value = age
	if = {
		limit = { has_variable = ac_youth_start_age }
		if = {
			limit = { age < ac_youth_end_age }
			value = age
			subtract = var:ac_youth_start_age
			multiply = 5
			divide = {
				value = ac_longevity_bonus
				min = 1
			}
			add = var:ac_youth_start_visual
		}
		else = {
			value = age
			subtract = ac_youth_end_age
			add = var:ac_youth_start_visual
			add = 5
		}
	}
	floor = yes
}
