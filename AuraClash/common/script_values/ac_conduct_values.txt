# Honor and Ruthless, as in Aura Clash: two 0-100 scales that start at 50.
ac_honor_value = { value = 50 if = { limit = { has_variable = ac_honor } value = var:ac_honor } }
ac_ruthless_value = { value = 50 if = { limit = { has_variable = ac_ruthless } value = var:ac_ruthless } }

# Style and Element in harmony with conduct speed Aura Pressure (the IF's "elemental sync").
ac_conduct_ap_harmony = {
	value = 0
	if = { limit = { has_trait = ac_style_akakiru ac_honor_value >= 70 } add = 0.1 }
	if = { limit = { has_trait = ac_style_aoyusumu ac_ruthless_value <= 30 } add = 0.1 }
	if = { limit = { has_trait = ac_style_kurokonton intrigue >= 15 } add = 0.1 }
	if = { limit = { has_trait = ac_style_kiihakai learning >= 15 } add = 0.1 }
	if = { limit = { has_trait = ac_style_midorikatai prowess >= 20 } add = 0.1 }
	if = { limit = { has_trait = ac_element_light ac_ruthless_value <= 35 } add = 0.05 }
	if = { limit = { has_trait = ac_element_fire ac_ruthless_value >= 75 } add = 0.05 }
	# Race (IF): Humans are made for cycling chi; Yokai take to Midorikatai; Oni to Ice and Fire.
	if = { limit = { has_trait = ac_race_human } add = 0.1 }
	if = { limit = { ac_is_yokai_trigger = yes has_trait = ac_style_midorikatai } add = 0.1 }
	if = { limit = { has_trait = ac_race_oni OR = { has_trait = ac_element_ice has_trait = ac_element_fire } } add = 0.05 }
}
