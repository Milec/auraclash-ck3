# Cheap learner checks run before looking through personal connections.
ac_ai_awakening_ready_trigger = {
 is_ai = yes
 is_alive = yes
 is_adult = yes
 is_imprisoned = no
 NOT = { has_trait = ac_cultivator }
 NOT = { has_trait = incapable }
 NOT = { ac_temporarily_unwell_trigger = yes }
 health >= 3
 ac_ai_cultivation_aspirant_trigger = yes
 ac_awakening_ready_trigger = yes
}

# Learner scope; use the actual interaction validity, including costs/range/cooldowns.
ac_ai_can_request_cultivation_trigger = {
 ac_ai_awakening_ready_trigger = yes
 prestige >= 200
 is_character_interaction_valid = {
  interaction = ac_request_cultivation_interaction
  recipient = $TEACHER$
 }
}

# A search should leave a reserve after paying 100 gold / 250 prestige / 250 piety.
ac_ai_can_seek_cultivation_trigger = {
 ac_ai_awakening_ready_trigger = yes
 is_ruler = yes
 has_trait = ambitious
 gold >= 200
 prestige >= 500
 piety >= 500
 can_execute_decision = ac_seek_cultivation_decision
}

# Offer training to interested AI friends/wards/partners, never force an ordeal on a player.
ac_ai_can_offer_cultivation_trigger = {
 is_alive = yes
 has_trait = ac_cultivator
 is_adult = yes
 is_imprisoned = no
 NOT = { has_trait = incapable }
 NOT = { ac_temporarily_unwell_trigger = yes }
 health >= 3
 prestige >= 500
 piety >= 500
}
