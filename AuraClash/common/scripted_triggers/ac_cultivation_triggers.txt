# Shared cooldown between studying a Style and attuning an Element
ac_path_study_ready_trigger = {
	custom_tooltip = {
		text = ac_path_study_cooldown_tt
		NOT = { has_character_flag = ac_path_study_cooldown }
	}
}

# Can this character run a Mortal Core Training on $CORE$ right now?
ac_can_train_core_trigger = {
	has_trait = ac_cultivator
	age >= 16
	custom_tooltip = {
		text = ac_core_training_cooldown_tt
		NOT = { has_character_flag = ac_core_training_cooldown }
	}
	prestige >= 100
	custom_tooltip = {
		text = ac_core_maxed_tt
		NOT = { ac_core_$CORE$_maxed_trigger = yes }
	}
}
