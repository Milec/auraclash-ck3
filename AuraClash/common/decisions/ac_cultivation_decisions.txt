# Aura Clash: Cultivation decisions

### Seek Cultivation: an ambitious outsider finances their own search ###
ac_seek_cultivation_decision = {
	decision_group_type = ac_cultivation
	picture = { reference = "gfx/interface/illustrations/decisions/decision_destiny_goal.dds" }
	desc = ac_seek_cultivation_decision_desc
	selection_tooltip = ac_seek_cultivation_decision_tooltip
	confirm_text = ac_seek_cultivation_decision_confirm
	ai_check_interval = 0 # Annual everyone pulse executes this decision after checking teachers.
	cooldown = { years = 1 }
	is_shown = { NOT = { has_trait = ac_cultivator } has_trait = ambitious }
	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
		NOT = { has_trait = incapable }
		has_trait = ambitious
		ac_awakening_ready_trigger = yes
	}
	cost = { gold = 100 prestige = 250 piety = 250 }
	effect = { ac_attempt_awakening_effect = yes }
	ai_potential = {
		gold >= 200 prestige >= 500 piety >= 500
		ac_ai_awakening_ready_trigger = yes
	}
	ai_will_do = { base = 100 } # The annual pulse supplies the willingness roll.
}
