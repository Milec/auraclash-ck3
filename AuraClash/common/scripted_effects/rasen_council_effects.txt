# "All five maintain shaky peace through a seat on the Exalted Council that rules from the capital."
# Global scope, yearly.
rasen_council_yearly_effect = {
	if = {
		limit = { NOT = { has_global_variable = rasen_council_countdown } }
		set_global_variable = { name = rasen_council_countdown value = 2 }
	}
	change_global_variable = { name = rasen_council_countdown add = -1 }
	if = {
		limit = { global_var:rasen_council_countdown <= 0 }
		set_global_variable = { name = rasen_council_countdown value = 3 }
		rasen_council_session_effect = yes
	}
	# Patronage is checked after the Trials (base mod) have had their own yearly pulse.
	if = { limit = { exists = title:k_azuma.holder } title:k_azuma.holder = { trigger_event = { id = rasen_council.0080 days = 45 } } }
	else_if = { limit = { rasen_seat_filled_trigger = { LAND = bladed } } global_var:rasen_exalt_bladed = { trigger_event = { id = rasen_council.0080 days = 45 } } }
	else_if = { limit = { rasen_seat_filled_trigger = { LAND = sacred } } global_var:rasen_exalt_sacred = { trigger_event = { id = rasen_council.0080 days = 45 } } }
	else_if = { limit = { rasen_seat_filled_trigger = { LAND = mystic } } global_var:rasen_exalt_mystic = { trigger_event = { id = rasen_council.0080 days = 45 } } }
	else_if = { limit = { rasen_seat_filled_trigger = { LAND = cursed } } global_var:rasen_exalt_cursed = { trigger_event = { id = rasen_council.0080 days = 45 } } }
	else_if = { limit = { rasen_seat_filled_trigger = { LAND = wild } } global_var:rasen_exalt_wild = { trigger_event = { id = rasen_council.0080 days = 45 } } }
}

# Global scope. Draw the decree in proportion to the votes and enact it.
rasen_council_enact_effect = {
	random_list = {
		0 = {
			modifier = { add = global_var:rasen_votes_bladed }
			set_global_variable = { name = rasen_decree value = flag:bladed }
		}
		0 = {
			modifier = { add = global_var:rasen_votes_sacred }
			set_global_variable = { name = rasen_decree value = flag:sacred }
		}
		0 = {
			modifier = { add = global_var:rasen_votes_mystic }
			set_global_variable = { name = rasen_decree value = flag:mystic }
		}
		0 = {
			modifier = { add = global_var:rasen_votes_cursed }
			set_global_variable = { name = rasen_decree value = flag:cursed }
		}
		0 = {
			modifier = { add = global_var:rasen_votes_wild }
			set_global_variable = { name = rasen_decree value = flag:wild }
		}
		# Nobody voted: the Bladed Exalt's decree stands by default.
		1 = {
			modifier = { factor = 0 OR = {
				global_var:rasen_votes_bladed > 0
				global_var:rasen_votes_sacred > 0
				global_var:rasen_votes_mystic > 0
				global_var:rasen_votes_cursed > 0
				global_var:rasen_votes_wild > 0
			} }
			set_global_variable = { name = rasen_decree value = flag:bladed }
		}
	}
	if = {
		limit = { global_var:rasen_decree = flag:bladed }
		rasen_decree_archons_effect = yes
		if = { limit = { rasen_seat_filled_trigger = { LAND = bladed } } global_var:rasen_exalt_bladed = { save_scope_as = rasen_council_winner } }
	}
	else_if = {
		limit = { global_var:rasen_decree = flag:sacred }
		rasen_decree_fiends_effect = yes
		if = { limit = { rasen_seat_filled_trigger = { LAND = sacred } } global_var:rasen_exalt_sacred = { save_scope_as = rasen_council_winner } }
	}
	else_if = {
		limit = { global_var:rasen_decree = flag:mystic }
		rasen_decree_archives_effect = yes
		if = { limit = { rasen_seat_filled_trigger = { LAND = mystic } } global_var:rasen_exalt_mystic = { save_scope_as = rasen_council_winner } }
	}
	else_if = {
		limit = { global_var:rasen_decree = flag:cursed }
		rasen_decree_levy_effect = yes
		if = { limit = { rasen_seat_filled_trigger = { LAND = cursed } } global_var:rasen_exalt_cursed = { save_scope_as = rasen_council_winner } }
	}
	else_if = {
		limit = { global_var:rasen_decree = flag:wild }
		rasen_decree_trials_effect = yes
		if = { limit = { rasen_seat_filled_trigger = { LAND = wild } } global_var:rasen_exalt_wild = { save_scope_as = rasen_council_winner } }
	}
	# The winning Exalt gains standing. Exalts outvoted on their own decree resent the winner.
	scope:rasen_council_winner ?= { add_prestige = 100 }
	if = {
		limit = { exists = scope:rasen_council_winner rasen_seat_filled_trigger = { LAND = bladed } NOT = { global_var:rasen_decree = flag:bladed }
			global_var:rasen_votes_bladed > 0 NOT = { global_var:rasen_exalt_bladed = scope:rasen_council_winner } }
		global_var:rasen_exalt_bladed = { add_opinion = { target = scope:rasen_council_winner modifier = rasen_council_outvoted_opinion } }
	}
	if = {
		limit = { exists = scope:rasen_council_winner rasen_seat_filled_trigger = { LAND = sacred } NOT = { global_var:rasen_decree = flag:sacred }
			global_var:rasen_votes_sacred > 0 NOT = { global_var:rasen_exalt_sacred = scope:rasen_council_winner } }
		global_var:rasen_exalt_sacred = { add_opinion = { target = scope:rasen_council_winner modifier = rasen_council_outvoted_opinion } }
	}
	if = {
		limit = { exists = scope:rasen_council_winner rasen_seat_filled_trigger = { LAND = mystic } NOT = { global_var:rasen_decree = flag:mystic }
			global_var:rasen_votes_mystic > 0 NOT = { global_var:rasen_exalt_mystic = scope:rasen_council_winner } }
		global_var:rasen_exalt_mystic = { add_opinion = { target = scope:rasen_council_winner modifier = rasen_council_outvoted_opinion } }
	}
	if = {
		limit = { exists = scope:rasen_council_winner rasen_seat_filled_trigger = { LAND = cursed } NOT = { global_var:rasen_decree = flag:cursed }
			global_var:rasen_votes_cursed > 0 NOT = { global_var:rasen_exalt_cursed = scope:rasen_council_winner } }
		global_var:rasen_exalt_cursed = { add_opinion = { target = scope:rasen_council_winner modifier = rasen_council_outvoted_opinion } }
	}
	if = {
		limit = { exists = scope:rasen_council_winner rasen_seat_filled_trigger = { LAND = wild } NOT = { global_var:rasen_decree = flag:wild }
			global_var:rasen_votes_wild > 0 NOT = { global_var:rasen_exalt_wild = scope:rasen_council_winner } }
		global_var:rasen_exalt_wild = { add_opinion = { target = scope:rasen_council_winner modifier = rasen_council_outvoted_opinion } }
	}
	every_player = { trigger_event = rasen_council.0002 }
}

rasen_decree_archons_effect = {
	every_county_in_region = {
		region = world_rasen_outer
		add_county_modifier = { modifier = rasen_archon_patrols_modifier years = 3 }
	}
	if = {
		limit = { has_global_variable = rasen_ice_incursion_active global_var:rasen_ice_remaining > 1 }
		change_global_variable = { name = rasen_ice_remaining add = -1 }
	}
}

rasen_decree_fiends_effect = {
	every_county_in_region = { region = world_rasen_inner add_county_modifier = { modifier = rasen_fiend_purge_modifier years = 3 } }
	every_county_in_region = { region = world_rasen_outer add_county_modifier = { modifier = rasen_fiend_purge_modifier years = 3 } }
	every_ruler = { limit = { faith = faith:devanic_temple } add_piety = 100 }
}

# Read by ac_path_manual_gold in the base mod.
rasen_decree_archives_effect = {
	set_global_variable = { name = rasen_archives_open value = yes years = 3 }
}

rasen_decree_levy_effect = {
	every_county_in_region = { region = world_rasen_outer add_county_modifier = { modifier = rasen_imperial_levy_modifier years = 3 } }
	if = { limit = { rasen_seat_filled_trigger = { LAND = bladed } } global_var:rasen_exalt_bladed = { add_gold = 150 } }
	if = { limit = { rasen_seat_filled_trigger = { LAND = sacred } } global_var:rasen_exalt_sacred = { add_gold = 150 } }
	if = { limit = { rasen_seat_filled_trigger = { LAND = mystic } } global_var:rasen_exalt_mystic = { add_gold = 150 } }

# The Trials open on the next yearly pulse.
rasen_decree_trials_effect = {
	if = {
		limit = { OR = { NOT = { has_global_variable = ac_ascension_countdown } global_var:ac_ascension_countdown > 1 } }
		set_global_variable = { name = ac_ascension_countdown value = 1 }
	}
}

# Global scope, 45 days after a yearly pulse. When the Trials have just opened,
# "the Exalts themselves are sending customary gifts to their favored among the Top Five".
rasen_trial_patronage_effect = {
	if = {
		limit = {
			has_global_variable = ac_ascension_countdown
			global_var:ac_ascension_countdown = ac_ascension_interval_years
			NOT = { has_global_variable = rasen_patronage_this_trials }
		}
		set_global_variable = { name = rasen_patronage_this_trials value = yes years = 1 }
		# One AI Exalt sponsors the favourite, tilting the scales in their land's eyes.
		if = {
			limit = { has_global_variable = ac_ascension_champion }
			random_list = {
				1 = {
					modifier = { factor = 0 NOT = { rasen_seat_filled_trigger = { LAND = bladed } } }
					modifier = { factor = 0 rasen_seat_filled_trigger = { LAND = bladed } global_var:rasen_exalt_bladed = { is_ai = no } }
					set_global_variable = { name = rasen_trial_patron_land value = flag:bladed }
					if = { limit = { rasen_seat_filled_trigger = { LAND = bladed } } global_var:rasen_exalt_bladed = { save_scope_as = rasen_trial_patron } }
				}
				1 = {
					modifier = { factor = 0 NOT = { rasen_seat_filled_trigger = { LAND = sacred } } }
					modifier = { factor = 0 rasen_seat_filled_trigger = { LAND = sacred } global_var:rasen_exalt_sacred = { is_ai = no } }

# Character scope. Take the seat of $LAND$: give up any lands, move to the High Steward's court and
# found a Celestial noble family with an estate on Azuma Honsha.
rasen_seat_exalt_effect = {
	save_scope_as = rasen_new_exalt
	if = {
		limit = { has_global_variable = rasen_exalt_$LAND$ global_var:rasen_exalt_$LAND$ = { is_alive = yes } }
		global_var:rasen_exalt_$LAND$ = { remove_variable = rasen_exalt_seat remove_variable = rasen_exalt_since }
	}
	set_global_variable = { name = rasen_exalt_$LAND$ value = scope:rasen_new_exalt }
	set_variable = { name = rasen_exalt_seat value = flag:$LAND$ }
	set_variable = { name = rasen_exalt_since value = current_year }
	if = {
		limit = { is_ruler = no exists = title:h_china.holder NOT = { liege_or_court_owner ?= title:h_china.holder } }
		title:h_china.holder = { add_courtier = scope:rasen_new_exalt }
	}
	if = {
		limit = { is_ruler = no exists = title:h_china.holder }
		create_noble_family_effect = { GOVERNMENT_GIVER = title:h_china.holder }
	}
}
