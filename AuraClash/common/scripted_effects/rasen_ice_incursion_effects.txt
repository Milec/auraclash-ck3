# "The Ice Incursions are a primordial, ever-present plague": malignant ice and yin spirits
# form demonic elemental hordes that invade Outer North every generation, followed by
# twenty to thirty years of calm. The bookmark opens just after a victorious war.
rasen_ice_incursion_yearly_effect = {
	if = {
		limit = { NOT = { has_global_variable = rasen_ice_countdown } }
		set_global_variable = { name = rasen_ice_countdown value = 20 }
	}
	if = {
		limit = { has_global_variable = rasen_ice_incursion_active }
		change_global_variable = { name = rasen_ice_remaining add = -1 }
		if = {
			limit = { global_var:rasen_ice_remaining <= 0 }
			rasen_end_ice_incursion_effect = yes
		}
	}
	else = {
		change_global_variable = { name = rasen_ice_countdown add = -1 }
		if = {
			limit = { global_var:rasen_ice_countdown <= 0 }
			rasen_begin_ice_incursion_effect = yes
		}
	}
}

rasen_begin_ice_incursion_effect = {
	set_global_variable = { name = rasen_ice_incursion_active value = yes }
	random_list = {
		1 = { set_global_variable = { name = rasen_ice_remaining value = 3 } }
		1 = { set_global_variable = { name = rasen_ice_remaining value = 4 } }
		1 = { set_global_variable = { name = rasen_ice_remaining value = 5 } }
		1 = { set_global_variable = { name = rasen_ice_remaining value = 6 } }
	}
	every_county_in_region = {
		region = world_rasen_outer_north
		add_county_modifier = { modifier = rasen_ice_incursion_county_modifier years = 6 }
	}
	every_player = {
		limit = { rasen_in_outer_north_trigger = yes }
		trigger_event = rasen_ice.0001
	}
	every_player = {
		limit = { has_trait = ac_archonate_disciple NOT = { rasen_in_outer_north_trigger = yes } }
		trigger_event = rasen_ice.0002
	}
}

rasen_end_ice_incursion_effect = {
	remove_global_variable = rasen_ice_incursion_active
	random_list = {
		1 = { set_global_variable = { name = rasen_ice_countdown value = 20 } }
		1 = { set_global_variable = { name = rasen_ice_countdown value = 25 } }
		1 = { set_global_variable = { name = rasen_ice_countdown value = 30 } }
	}
	every_county_in_region = {
		region = world_rasen_outer_north
		remove_county_modifier = rasen_ice_incursion_county_modifier
	}
	every_player = {
		limit = { rasen_in_outer_north_trigger = yes }
		trigger_event = rasen_ice.0020
	}
}

# Character scope: march to the front for a season.
rasen_go_to_ice_front_effect = {
	add_character_flag = { flag = rasen_at_ice_front months = 6 }
	custom_tooltip = rasen_ice_front_tt
	hidden_effect = { trigger_event = { id = rasen_ice.0010 days = 120 } }
}
