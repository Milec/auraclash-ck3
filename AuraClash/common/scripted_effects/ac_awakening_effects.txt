# Requests, offers of training and the paid search share an attempt cooldown.
ac_attempt_awakening_effect = {
	add_character_flag = { flag = ac_awakening_cooldown years = 1 }
	custom_tooltip = ac_awaken_cultivation_decision_check_tt
	random_list = {
		0 = {
			modifier = { add = ac_awaken_success_chance }
			desc = ac_awaken_cultivation_decision.success
			send_interface_toast = {
				type = event_toast_effect_good
				title = ac_awaken_success_toast
				left_icon = this
				add_trait = ac_cultivator
			}
			ac_update_visual_age_effect = yes
			ac_offer_foundation_effect = yes
		}
		0 = {
			modifier = { add = ac_awaken_failure_chance }
			desc = ac_awaken_cultivation_decision.failure
			send_interface_toast = {
				type = event_toast_effect_bad
				title = ac_awaken_failure_toast
				left_icon = this
				increase_wounds_no_death_effect = { REASON = fight }
			}
		}
	}
}

# Resolve against the person selected in the character interaction.
ac_request_cultivation_effect = {
	if = {
		limit = { ac_can_request_cultivation_trigger = { TEACHER = $TEACHER$ } }
		if = {
			limit = { NOT = { ac_has_willing_training_connection_trigger = { TEACHER = $TEACHER$ } } }
			use_hook = $TEACHER$
		}
		ac_attempt_awakening_effect = yes
	}
}

# Store the actual guardian, so earlier teaching cannot qualify strangers.
ac_record_guardian_teacher_effect = {
	add_character_flag = ac_cultivator_guardian_taught
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = ac_cultivation_guardians target = $TEACHER$ } } }
		add_to_variable_list = { name = ac_cultivation_guardians target = $TEACHER$ }
	}
}

ac_record_cultivator_guardian_effect = {
	every_relation = {
		type = guardian
		limit = { ac_cultivation_teacher_trigger = yes }
		save_scope_as = ac_recorded_guardian
		prev = { ac_record_guardian_teacher_effect = { TEACHER = scope:ac_recorded_guardian } }
	}
}
