# Spend prestige to raise one Combat Core (a track on the Cultivator trait)
ac_train_core_effect = {
	add_prestige = -100
	add_trait_xp = {
		trait = ac_cultivator
		track = ac_core_$CORE$
		value = 5
	}
	add_character_flag = {
		flag = ac_core_training_cooldown
		years = 1
	}
	set_variable = {
		name = ac_core_last
		value = flag:$CORE$
	}
	if = {
		limit = { has_character_flag = ac_core_training_auto }
		custom_tooltip = ac_core_auto_will_repeat_tt
		hidden_effect = {
			trigger_event = {
				id = ac_cult.0002
				days = 366
			}
		}
	}
}

# Called by the quarterly pulse and by the scheduled ac_cult.0002 check
ac_core_auto_train_effect = {
	if = {
		limit = {
			has_trait = ac_cultivator
			has_character_flag = ac_core_training_auto
			has_variable = ac_core_last
		}
		if = {
			limit = {
				var:ac_core_last = flag:soul
				ac_can_train_core_trigger = { CORE = soul }
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = ac_core_auto_trained_toast
				left_icon = root
				ac_train_core_effect = { CORE = soul }
			}
		}
		else_if = {
			limit = {
				var:ac_core_last = flag:agility
				ac_can_train_core_trigger = { CORE = agility }
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = ac_core_auto_trained_toast
				left_icon = root
				ac_train_core_effect = { CORE = agility }
			}
		}
		else_if = {
			limit = {
				var:ac_core_last = flag:power
				ac_can_train_core_trigger = { CORE = power }
			}
			send_interface_toast = {
				type = event_toast_effect_good
				title = ac_core_auto_trained_toast
				left_icon = root
				ac_train_core_effect = { CORE = power }
			}
		}
		else_if = {
			limit = {
				var:ac_core_last = flag:fortitude
				ac_can_train_core_trigger = { CORE = fortitude }

			send_interface_toast = {

# One random Core dot (+5 XP) on a Core that still has room
ac_add_random_core_dot_effect = {
	if = {
		limit = {
			OR = {
				ac_core_soul_maxed_trigger = no
				ac_core_agility_maxed_trigger = no
				ac_core_power_maxed_trigger = no
				ac_core_fortitude_maxed_trigger = no
			}
		}
		random_list = {
			10 = {
				trigger = { ac_core_soul_maxed_trigger = no }
				add_trait_xp = { trait = ac_cultivator track = ac_core_soul value = 5 }
			}
			10 = {
				trigger = { ac_core_agility_maxed_trigger = no }
				add_trait_xp = { trait = ac_cultivator track = ac_core_agility value = 5 }
			}
			10 = {
				trigger = { ac_core_power_maxed_trigger = no }
				add_trait_xp = { trait = ac_cultivator track = ac_core_power value = 5 }
			}
			10 = {
				trigger = { ac_core_fortitude_maxed_trigger = no }
				add_trait_xp = { trait = ac_cultivator track = ac_core_fortitude value = 5 }
			}
		}
	}
}

# Fame traits reward existing cultivators with $DOTS$ Core dots, once per trait
ac_fame_cultivation_effect = {
	if = {
		limit = {
			has_trait = ac_cultivator
			has_trait = $TRAIT$
			NOT = { has_character_flag = ac_fame_gift_$TRAIT$ }
		}
		add_character_flag = ac_fame_gift_$TRAIT$
		send_interface_toast = {
			type = event_toast_effect_good
			title = ac_fame_cultivation_toast
			left_icon = root
			while = {
				count = $DOTS$
				ac_add_random_core_dot_effect = yes
			}
		}
		hidden_effect = {
			ac_update_visual_age_effect = yes
			ac_offer_foundation_effect = yes
		}
	}
}

			}

		}
	}
}

# Record when the slowed-aging phase begins (no retroactive rejuvenation)
ac_start_youth_effect = {
	if = {
		limit = {
			has_trait = ac_cultivator
			age >= 22
			NOT = { has_variable = ac_youth_start_age }
		}
		set_variable = {
			name = ac_youth_start_age
			value = age
		}
		if = {
			limit = { age <= 22 }
			set_variable = {
				name = ac_youth_start_visual
				value = 22
			}
		}
		else = {
			set_variable = {
				name = ac_youth_start_visual
				value = age
			}
		}
	}
}

# Health watch: while Preserved Youth is active, a hidden check (ac_cult.0400) runs every
# 30 days when health is below 3, otherwise every 90 days, and stops when the trait goes.
ac_start_health_watch_effect = {
	if = {
		limit = {
			has_trait = ac_preserved_youth
			NOT = { has_character_flag = ac_health_watch }
		}
		add_character_flag = {
			flag = ac_health_watch
			days = 100
		}
		trigger_event = {
			id = ac_cult.0400
			days = 30
		}
	}
}

# Keep the portrait young (game rule "Cultivator Aging": Slowed)
ac_update_visual_age_effect = {
	ac_start_youth_effect = yes
	if = {
		limit = {
			has_game_rule = ac_cultivator_aging_slowed
			has_trait = ac_cultivator
			has_variable = ac_youth_start_age
			age < ac_youth_release_age
			OR = {
				health >= 1
				ac_temporarily_unwell_trigger = yes
			}
		}
		if = {
			limit = { NOT = { has_trait = ac_preserved_youth } }
			add_trait = ac_preserved_youth
		}
		set_immortal_age = ac_visual_age
		ac_start_health_watch_effect = yes
	}
	else_if = {
		limit = { has_trait = ac_preserved_youth }
		send_interface_toast = {
			type = event_toast_effect_bad
			title = ac_death_spiral_toast
			left_icon = root
			remove_trait = ac_preserved_youth
		}
	}
}

# AI: the whole yearly cultivation routine (one call per adult AI cultivator per year)
ac_ai_yearly_cultivation_effect = {
	# Foundation
	if = {
		limit = { ac_has_foundation_trigger = no }
		ac_random_foundation_effect = yes
	}

	# Mortal Core Training: random unmaxed Core
	if = {
		limit = {
			prestige > 300
			NOT = { has_character_flag = ac_core_training_cooldown }
			OR = {
				ac_core_soul_maxed_trigger = no
				ac_core_agility_maxed_trigger = no
				ac_core_power_maxed_trigger = no
				ac_core_fortitude_maxed_trigger = no
			}
		}
		random_list = {
		10 = {
			trigger = { ac_core_soul_maxed_trigger = no }
			ac_train_core_effect = { CORE = soul }
		}
		10 = {
			trigger = { ac_core_agility_maxed_trigger = no }
			ac_train_core_effect = { CORE = agility }
		}
		10 = {
			trigger = { ac_core_power_maxed_trigger = no }
			ac_train_core_effect = { CORE = power }
		}
		10 = {
			trigger = { ac_core_fortitude_maxed_trigger = no }
			ac_train_core_effect = { CORE = fortitude }
		}
		}
	}

	# Style or Element (shared yearly cooldown)
	if = {
		limit = { NOT = { has_character_flag = ac_path_study_cooldown } }
		if = {
			limit = {
				ac_has_martial_style_trigger = no
				prestige >= 150
			}
			random = {
				chance = 15
				add_prestige = -150
				add_character_flag = {
					flag = ac_path_study_cooldown
					years = 1
				}
				random_list = {
		10 = {
			modifier = { add = stewardship }
			ac_attempt_style_effect = { STYLE = akakiru }
		}
		10 = {
			modifier = { add = diplomacy }
			ac_attempt_style_effect = { STYLE = aoyusumu }
		}
		10 = {
			modifier = { add = learning }
			ac_attempt_style_effect = { STYLE = kiihakai }
		}
		10 = {
			modifier = { add = intrigue }
			ac_attempt_style_effect = { STYLE = kurokonton }
		}
		10 = {
			modifier = { add = martial }
			ac_attempt_style_effect = { STYLE = midorikatai }
		}
				}
			}
		}
		else_if = {
			limit = {
				ac_element_count < 2
				piety >= 100
			}
			random = {
				chance = 15
				add_piety = -100
				add_character_flag = {
					flag = ac_path_study_cooldown
					years = 1
				}
				random_list = {
		10 = {
			trigger = { NOT = { has_trait = ac_element_darkness } }
			modifier = { add = ac_learn_element_darkness_chance }
			ac_attempt_element_effect = { ELEMENT = darkness }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_earth } }
			modifier = { add = ac_learn_element_earth_chance }
			ac_attempt_element_effect = { ELEMENT = earth }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_fire } }
			modifier = { add = ac_learn_element_fire_chance }
			ac_attempt_element_effect = { ELEMENT = fire }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_ice } }
			modifier = { add = ac_learn_element_ice_chance }
			ac_attempt_element_effect = { ELEMENT = ice }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_light } }
			modifier = { add = ac_learn_element_light_chance }
			ac_attempt_element_effect = { ELEMENT = light }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_storm } }
			modifier = { add = ac_learn_element_storm_chance }
			ac_attempt_element_effect = { ELEMENT = storm }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_water } }
			modifier = { add = ac_learn_element_water_chance }
			ac_attempt_element_effect = { ELEMENT = water }
		}
		10 = {
			trigger = { NOT = { has_trait = ac_element_wind } }
			modifier = { add = ac_learn_element_wind_chance }
			ac_attempt_element_effect = { ELEMENT = wind }
		}
				}
			}
		}
	}

	# Immortal Core Advancement
	if = {
		limit = {
			NOT = { has_trait = ac_immortal_core }
			NOT = { has_character_flag = ac_immortal_attempt_cooldown }
			piety >= 150
			ac_martial_plus_learning >= 16
		}
		random = {
			chance = 30
			add_piety = -150
			add_character_flag = {
				flag = ac_immortal_attempt_cooldown
				years = 1
			}
			ac_immortal_core_attempt_effect = yes
		}
	}

	# Immortal Core experience: AI banks it in a variable and unlocks a perk per
	# 1000 XP, so it never depends on the AI choosing a Cultivation focus
	if = {
		limit = { has_trait = ac_immortal_core }
		if = {
			limit = { NOT = { has_variable = ac_ai_cultivation_xp } }
			set_variable = {
				name = ac_ai_cultivation_xp
				value = 0
			}
		}
		change_variable = {
			name = ac_ai_cultivation_xp
			add = 240
		}
		if = {
			limit = { var:ac_ai_cultivation_xp >= 1000 }
			change_variable = {
				name = ac_ai_cultivation_xp
				add = -1000
			}
			ac_ai_grant_cultivation_perk_effect = yes
		}
	}
}
