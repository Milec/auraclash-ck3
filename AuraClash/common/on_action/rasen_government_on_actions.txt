# The merit bureaucracy expects every governor to head a noble family with a family title and estate.
# History gives the Magistrates and the High Steward none, so they are founded when the game starts.
on_game_start_after_lobby = { on_actions = { rasen_found_noble_families } }
rasen_found_noble_families = {
	effect = {
		every_ruler = {
			limit = {
				has_government = celestial_government
				highest_held_title_tier >= tier_county
				NOT = { any_held_title = { is_noble_family_title = yes } }
			}
			create_noble_family_effect = { GOVERNMENT_GIVER = this }
		}
	}
}

# CK3 1.21 gives Soryo (japan_feudal) rulers a starting succession law only when they are
# independent or carry the plain feudal flag; the Bladed Lands answer to the hegemony, so
# their rulers started with none ("unhandled succession order [invalid]"). Primogeniture.
on_game_start = { on_actions = { rasen_soryo_succession } }
rasen_soryo_succession = {
	effect = {
		every_ruler = {
			limit = {
				government_has_flag = government_is_japan_feudal
				NOR = {
					has_realm_law = confederate_partition_succession_law
					has_realm_law = partition_succession_law
					has_realm_law = high_partition_succession_law
					has_realm_law = single_heir_succession_law
					has_realm_law = single_heir_succession_law_youngest
					has_realm_law = single_heir_dynasty_house
				}
			}
			add_realm_law_skip_effects = single_heir_succession_law
		}
	}
}
