# Appended to vanilla on_actions (lists merge; nothing vanilla is replaced).
# Cultivation routines use cheap eligibility checks before doing further work.
quarterly_playable_pulse = {
	on_actions = { ac_quarterly_player_pulse ac_fame_cultivation_pulse }
}
random_yearly_everyone_pulse = {
	on_actions = { ac_yearly_ai_pulse ac_fame_cultivation_pulse ac_guardian_teaching_pulse }
}
on_birthday = {
	on_actions = { ac_birthday_pulse ac_guardian_teaching_pulse }
}
on_game_start_after_lobby = {
	on_actions = { ac_game_start }
}
on_combat_end_winner = {
	on_actions = { ac_on_combat_end_winner }
}
on_combat_end_loser = {
	on_actions = { ac_on_combat_end_loser }
}

# Players: auto-training, Immortal Core XP, Foundation catch-up, youth health check
ac_quarterly_player_pulse = {
	trigger = {
		has_trait = ac_cultivator
		is_ai = no
	}
	effect = {
		ac_core_auto_train_effect = yes
		if = {
			limit = {
				has_trait = ac_immortal_core
				age >= 16
			}
			add_ac_cultivation_lifestyle_xp = 60
		}
		ac_offer_foundation_effect = yes
		if = {
			limit = { has_trait = ac_preserved_youth }
			ac_update_visual_age_effect = yes
		}
	}
}

# AI: one routine per adult cultivator per year, on a random day (spreads the load)
ac_yearly_ai_pulse = {
	trigger = {
		has_trait = ac_cultivator
		is_ai = yes
		age >= 16
	}
	effect = {
		ac_ai_yearly_cultivation_effect = yes
		ac_start_health_watch_effect = yes
	}
}

# Fame bonuses require cultivation; fame cannot bypass awakening access.
ac_fame_cultivation_pulse = {
	trigger = {
		has_trait = ac_cultivator
		OR = {
			has_trait = conqueror
			has_trait = despoiler_of_byzantium
			has_trait = greatest_of_khans
		}
	}
	effect = {
		ac_fame_cultivation_all_effect = yes
	}
}

# Everyone: portrait age, and the Foundation choice on reaching adulthood
ac_birthday_pulse = {
	trigger = { has_trait = ac_cultivator }
	effect = {
		ac_update_visual_age_effect = yes
		ac_offer_foundation_effect = yes
	}
}

# Game start: set up existing cultivators (runs once)
ac_game_start = {
	effect = {
		every_living_character = {
			limit = { age < 16 NOT = { has_trait = ac_cultivator } }
			ac_record_cultivator_guardian_effect = yes
		}
		every_living_character = {
			limit = {
				OR = {
					has_trait = conqueror
					has_trait = despoiler_of_byzantium
					has_trait = greatest_of_khans
				}
			}
			ac_fame_cultivation_all_effect = yes
		}
		every_living_character = {
			limit = { has_trait = ac_cultivator }
			ac_update_visual_age_effect = yes
			ac_offer_foundation_effect = yes
		}
	}
}

# root = combat side
ac_on_combat_end_winner = {
	effect = {
		every_side_commander = {
			ac_gain_foundation_xp_effect = { XP = 3 }
		}
		every_side_knight = {
			ac_gain_foundation_xp_effect = { XP = 2 }
		}
	}
}
ac_on_combat_end_loser = {
	effect = {
		every_side_commander = {
			ac_gain_foundation_xp_effect = { XP = 1 }
		}
		every_side_knight = {
			ac_gain_foundation_xp_effect = { XP = 1 }
		}
	}
}

# Relation callbacks cover both directions, including guardians appointed
# shortly before adulthood. Removal records a guardian who awakened later.
on_set_relation_guardian = { on_actions = { ac_ward_guardian_teaching } }
on_remove_relation_guardian = { on_actions = { ac_ward_guardian_teaching } }
on_set_relation_ward = { on_actions = { ac_guardian_ward_teaching } }
on_remove_relation_ward = { on_actions = { ac_guardian_ward_teaching } }
ac_ward_guardian_teaching = {
	trigger = { age <= 16 scope:target = { ac_cultivation_teacher_trigger = yes } }
	effect = { ac_record_guardian_teacher_effect = { TEACHER = scope:target } }
}
ac_guardian_ward_teaching = {
	trigger = { ac_cultivation_teacher_trigger = yes scope:target = { age <= 16 } }
	effect = { save_scope_as = ac_recorded_guardian scope:target = { ac_record_guardian_teacher_effect = { TEACHER = scope:ac_recorded_guardian } } }
}

# Catch guardians who themselves awaken after being appointed.
ac_guardian_teaching_pulse = {
	trigger = { age <= 16 NOT = { has_trait = ac_cultivator } }
	effect = { ac_record_cultivator_guardian_effect = yes }
}
