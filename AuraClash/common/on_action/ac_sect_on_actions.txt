on_game_start_after_lobby = { on_actions = { ac_sect_initialize } }
# Existing saves are initialized lazily on the next playable pulse as well.
quarterly_playable_pulse = { on_actions = { ac_sect_initialize ac_sect_trial_repair } }
on_death = { on_actions = { ac_sect_member_death } }
ac_sect_initialize = {
	trigger = { NOT = { has_global_variable = ac_sects_initialized } }
	effect = {
		set_global_variable = { name = ac_sects_initialized value = 1 }
		every_kingdom = {
			limit = { ac_sect_kingdom_eligible_trigger = yes }
			save_scope_as = ac_sect_kingdom
			title_capital_county.title_province = { save_scope_as = ac_sect_hq }
			title_capital_county.holder = {
				create_character = {
					gender_female_chance = 50
					age = 45
					culture = this.culture
					faith = this.faith
					location = scope:ac_sect_hq
					random_traits = yes
					save_scope_as = ac_founder
				}
			}
			scope:ac_founder = {
				add_gold = 100
				ac_sect_found_effect = yes
				ac_sect_seed_roster_effect = yes
			}
		}
		every_in_global_list = { variable = ac_sects ac_sect_ai_diplomacy_effect = yes }
	}
}
ac_sect_trial_repair = {
	trigger = { OR = { has_variable = ac_sect_trial_activity has_variable = ac_sect_trial_paid } }
	effect = {
		if = {
			limit = { NOT = { involved_activity ?= var:ac_sect_trial_activity } }
			ac_sect_refund_trial_effect = yes
		}
	}
}
ac_sect_member_death = {
	trigger = { has_variable = ac_sect }
	effect = {
		# Story owner death transfers leadership separately. Do not clear its scope first.
		if = {
			limit = { NOT = { var:ac_sect.story_owner = this } }
			ac_sect_leave_effect = yes
		}
	}
}
