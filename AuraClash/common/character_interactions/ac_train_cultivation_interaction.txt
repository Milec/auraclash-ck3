# Train in Cultivation: teach a child, ward, friend, partner, sect member or hook holder
ac_train_cultivation_interaction = {
	category = interaction_category_friendly
	common_interaction = yes
	icon = mentor
	desc = ac_train_cultivation_interaction_desc

	cooldown_against_recipient = { years = 1 }
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			has_trait = ac_cultivator
			age >= 16
		}
		scope:recipient = {
			NOT = { has_trait = ac_cultivator }
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = { is_imprisoned = no NOT = { has_trait = incapable } }
		scope:recipient = {
			is_imprisoned = no
			NOT = { has_trait = incapable }
			ac_awakening_ready_trigger = yes
			custom_tooltip = {
				text = ac_train_cultivation_relation_tt
				ac_can_receive_cultivation_training_trigger = { TEACHER = scope:actor }
			}
		}
	}

	cost = {
		piety = 250
		prestige = 250
	}

	on_accept = {
		scope:recipient = {
			add_character_flag = { flag = ac_awakening_cooldown years = 1 }
			if = {
				limit = { has_usable_hook = scope:actor NOT = { ac_has_willing_training_connection_trigger = { TEACHER = scope:actor } } }
				use_hook = scope:actor
			}
		}
		scope:actor = {
			custom_tooltip = ac_train_cultivation_chance_tt
			random_list = {
				0 = {
					modifier = { add = scope:recipient.ac_train_cultivation_success_chance }
					desc = ac_train_cultivation_interaction.success
					send_interface_toast = {
						type = event_toast_effect_good
						title = ac_train_cultivation_success_toast
						left_icon = scope:actor
						right_icon = scope:recipient
						scope:recipient = { add_trait = ac_cultivator }
					}
					scope:recipient = {
						hidden_effect = {
							ac_update_visual_age_effect = yes
							ac_offer_foundation_effect = yes
						}
					}
				}
				0 = {
					modifier = { add = scope:recipient.ac_train_cultivation_failure_chance }
					desc = ac_train_cultivation_interaction.failure
					send_interface_toast = {
						type = event_toast_effect_bad
						title = ac_train_cultivation_failure_toast
						left_icon = scope:actor
						right_icon = scope:recipient
						scope:recipient = { increase_wounds_no_death_effect = { REASON = fight } }
					}
				}
			}
		}
	}

	ai_frequency = 12
	ai_targets = { ai_recipients = scripted_relations }
	ai_targets = { ai_recipients = family }
	ai_targets = { ai_recipients = spouses }
	ai_targets = { ai_recipients = courtiers max = 10 }
	is_available = {
		has_trait = ac_cultivator
		is_adult = yes
		OR = { is_ai = no ac_ai_can_offer_cultivation_trigger = yes }
	}
	ai_will_do = {
		base = 25
		modifier = { add = 20 scope:actor = { has_trait = compassionate } }
		modifier = { add = 15 scope:actor = { has_trait = generous } }
		modifier = { add = 10 scope:actor = { has_trait = diligent } }
		modifier = { add = -20 scope:actor = { has_trait = greedy } }
		modifier = { add = -20 scope:actor = { has_trait = callous } }
		modifier = {
			factor = 0
			NOT = { scope:actor = { ac_ai_can_offer_cultivation_trigger = yes } }
		}
		modifier = {
			factor = 0
			NOT = { scope:recipient = { ac_ai_awakening_ready_trigger = yes } }
		}
		modifier = {
			factor = 0
			NOT = { scope:actor = { opinion = { target = scope:recipient value >= 25 } } }
		}
	}
}
