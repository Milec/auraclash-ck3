# Perk-unlocked cultivation techniques
# Scry and Hex are now Seals (tools/build_seals.py).
### Copy Technique (Kurokonton) ###
ac_copy_technique_interaction = {
	category = interaction_category_prison
	common_interaction = yes
	icon = prison
	desc = ac_copy_technique_interaction_desc

	cooldown = { years = 1 }
	auto_accept = yes

	is_shown = {
		NOT = { scope:actor = scope:recipient }
		scope:actor = {
			OR = {
			has_perk = ac_3_style_kurokonton_e_perk
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = ac_copy_technique_prisoner_tt
				is_imprisoned_by = scope:actor
			}
			custom_tooltip = {
				text = ac_copy_technique_education_tt
				OR = {
				OR = { has_education_martial_trigger = yes has_trait = education_martial_prowess_1 has_trait = education_martial_prowess_2 has_trait = education_martial_prowess_3 has_trait = education_martial_prowess_4 }
				has_education_diplomacy_trigger = yes
				has_education_stewardship_trigger = yes
				has_education_intrigue_trigger = yes
				has_education_learning_trigger = yes
				}
			}
		}
	}

	cost = {
		piety = 100
	}

	on_accept = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = ac_copy_technique_toast
				left_icon = scope:actor
				right_icon = scope:recipient
			if = {
				limit = { scope:recipient = { OR = { has_education_martial_trigger = yes has_trait = education_martial_prowess_1 has_trait = education_martial_prowess_2 has_trait = education_martial_prowess_3 has_trait = education_martial_prowess_4 } } }
				add_martial_lifestyle_perk_points = 1
			}
			else_if = {
				limit = { scope:recipient = { has_education_diplomacy_trigger = yes } }
				add_diplomacy_lifestyle_perk_points = 1
			}
			else_if = {
				limit = { scope:recipient = { has_education_stewardship_trigger = yes } }
				add_stewardship_lifestyle_perk_points = 1
			}
			else_if = {
				limit = { scope:recipient = { has_education_intrigue_trigger = yes } }
				add_intrigue_lifestyle_perk_points = 1
			}
			else_if = {
				limit = { scope:recipient = { has_education_learning_trigger = yes } }
				add_learning_lifestyle_perk_points = 1
			}
			}
		}
	}

	ai_will_do = { base = 0 }
}
